# 指针符号

int var = 20;  // 普通变量
int *p;        // 声明一个指针变量
p = &var;      // 把普通变量的地址赋给指针变量 
int **pp = &p; // 把指针变量的地址赋给指针变量

在声明时, *是指针声明符号，右边是指针变量名, 左边是指针指向的数据类型
在使用时, * 是“解引用操作符”，表示“访问指针所指向的对象”
    **pp 是“解两次引用”：
        第一次取出“指针变量 p”
        第二次取出“指针变量 p 指向的var值”
可以对指针进行四种算术运算：++、--、+、-
    这些运算的单位是“所指向类型的大小”，而不一定是字节。
        p + 1 == 地址 + sizeof(*p)
指针变量在条件中使用时
    NULL → 等价于 0, 假
    非 NULL → 非 0, 真
指针声明符号 * 前后 空格是可选的，编译器会正确识别
    int* p, q;
        这行代码中只有 p 是指针，q 是普通 int
    int *p, *q;
指针变量里只有一个地址值,不包含长度,类型信息,结束位置,边界检查
    指针只知道“从哪里开始”，而不知道“到哪里结束”

// printf 并不知道字符串多长，它只是：
while (*p != '\0') { // 长度是靠 '\0' 结束符约定的
    putchar(*p);
    p++;
}
 
// strlen(p) 同样
size_t strlen(const char *p) {
    size_t n = 0;
    while (p[n] != '\0') n++;
    return n;
}

数组之所以“看起来知道长度”
    int a[10];
        因为在编译期,编译器知道 a 有 10 个元素所以：
            sizeof(a) == 10 * sizeof(int)
    在运行期
        int *p = a;
            此时：a 退化成 int *,长度信息立即消失
            sizeof(p)  // 只是指针大小 (4 or 8)
    数组知道长度，指针不知道, 一旦退化，就永远丢失


char *strs[] = {"apple", "banana", "cherry"};
char **cp = strs;
printf("%s\n", cp[0]); // apple
printf("%s\n", cp[1]); // banana
printf("%s\n", cp[2]); // cherry

"apple"
    类型为char[6], 字符串字面量, 含有 'a','p','p','l','e',和结尾'\0'
strs
    类型为char *[3], “字符串指针数组”，即数组里放了三个 char * 指针变量(字符串字面量在表达式中自动退化)
char **cp = strs;
    此表达式中数组名 strs 自动退化为指向数组中第一个指针变量的指针 &strs[0]
    cp 里存放的是指针变量 strs[0] 的地址
    指针变量 strs[0]，指向字符串字面量 "apple" 的第一个字符 'a' 的地址
cp不是数组，它本身并不“拥有”下标。[] 不是数组专属的操作符，它其实是语法糖：
    a[i] == *(a + i)
    只要 a 是一个能做“加法寻址”的指针，就能用 a[i] 形式访问数据。
        在早期的 BCPL / B 语言中，数组访问就是 *(a + i)。但是可读性差，不直观。于是增加了 a[i] 这种形式，让程序员看起来更像“下标访问”。
        以下三种语法上完全等价, 编译器在编译阶段就把它展开了, 生成的机器码完全相同：
            x = a[i];
            x = *(a + i);
            x = i[a];  // 因为 a[i] ≡ *(a + i) ≡ *(i + a), 所以 i[a] 也合法 
    cp[0]
        实际上等价于：*(cp + 0) → *cp → 指针变量 strs[0]
            printf("%s", ...) 之所以“打印出整个字符串”，是因为 %s 的行为定义——它会把这个指针当作一个指向以 '\0' 结尾的字符序列的地址，从该地址开始连续打印字符，直到遇到 '\0'
    cp[1]
        cp[1]  ≡  *(cp + 1)
        cp + 1 表示指针“向后移动一个 char * 大小的地址”；也就是指向指针变量 strs[1] 的地址
        再 * 一下（解引用）→ 得到指针变量 strs[1]
*cp      → 得到指针变量 strs[0] 即 "apple" 的首地址, 指向 'a' 的地址
**cp     → 得到指针变量 strs[0] 所指的字符数组的首元素 'a'
*(cp+1)  → 得到指针变量 strs[1] 即 "banana" 的首地址, 指向 'b' 的地址
*(*(cp+1)+2) → "banana"[2] → 'n'
    *(cp + 1) + 2 
        现在是对字符串指针做指针运算。
            "banana" 是 char[7]（含 \0）。
            "banana" + 2 指向第 2 个索引的字符 'n' 的地址
    *(*(cp+1)+2)
        解前面指针运算的结果得到实际字符 'n'

# 悬空指针
指针变量指向的内存已经被释放或失效，但指针本身仍然保存着旧地址。访问它会导致 未定义行为（UB），可能崩溃或者得到垃圾值

int *p = malloc(sizeof(int));
if (p == NULL) return 1;
free(p);
p = NULL;  // ✅ 避免悬空
free(p);   // ✅ 安全！ free(NULL) 什么都不做
